
public class ServeClass1 {
	int foo = 0;

	/*public void initialize() {
		foo = 0;
	}*/

	public void counter(int numMax) {
		int count;
		
		if (numMax < 0)
		{
			return ;
		}

		count = 1;
		do {
			out.println("Value: ", count);
			count = count + 1;
		} while (count < numMax);
	}

	public int comparator(int num1, int num2) {
		int valRet;
		
		valRet = 1;
		out.println("Result: ");
	
		if (num1 == num2) {  // Show num1 == num2
			out.print("\t", num1);
			out.print("==");
			out.println("", num2);
			valRet = valRet + 1;
			foo = foo + (num1 + num2);
		} else if (num1 != num2) {  // Show num1 != num2
			out.print("\t", num1);
			out.print("!=");
			out.println("", num2);
			foo = foo + (num1 * num2);
			valRet = valRet * -4;
		} else if (num1 < num2) {  // Show num1 < num2
			out.print("\t", num1);
			out.print("<");
			out.println("", num2);
			foo = foo + (num2 - num1);
			valRet = valRet + 100 / - (4 * 2);
		} else if (num1 > num2) {  // Show num1 > num2
			out.print("\t", num1);
			out.print(">");
			out.println("", num2);
			foo = foo + (num1 - num2);
			valRet = valRet / 100 * 2;
		}
		
		return valRet;
	}
}

public class ServeClass2 {
	private String name;
	private double points;
	private int level;
	
	public void clear() {
		name = "";
		points = 0.0d;
		level = 0;
	}

	public void fill(String n, double p, int l) {
		name = n;
		points = p;
		level = l;
	}
	
	public String getName() {
		return name;
	}
	
	public double getPoints() {
		return points;
	}
	
	public int getLevel() {
		return level;
	}
	
	public String getStr() {
		String info;
		
		switch(level) {
			case 1:
				info = "Basic";
			case 2:
				info = "Middle";
			case 3:
				info = "Advanced";
			default:
				info = "Unknown";
		}
		
		return info;
	}
}
